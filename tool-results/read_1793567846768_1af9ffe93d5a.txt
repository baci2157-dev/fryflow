    56→/** Яндекс.Карты поддерживают до 10 точек мультимаршрута (склад + 9). */
    57→const YANDEX_MAX_STOPS = 9;
    58→
    59→export default function RoutePage() {

    61→  const routeStops = useCrmStore((s) => s.routeStops);
    62→  const capacity = useCrmStore((s) => s.capacity);
    63→  const setCapacity = useCrmStore((s) => s.setCapacity);
    64→  const setRouteStops = useCrmStore((s) => s.setRouteStops);
    65→  const setStopBoxes = useCrmStore((s) => s.setStopBoxes);
    66→  const removeRouteStop = useCrmStore((s) => s.removeRouteStop);

   114→        title: "Яндекс ограничивает маршрут 10 точками",

   120→  };
   121→
   122→  /* ── В дороге: действия по точке ── */
   123→  const currentStop = plan.ordered.find((s) => !flags[s.client.id]?.done) ?? null;
   124→
   125→  const markStop = (

   127→    kind: "delivered" | "unpaid" | "noshow",
   128→    boxes: number

   130→    setFlags((f) => ({
   131→      ...f,
   132→      [client.id]: {
   133→        done: true,
   134→        unpaid: kind === "unpaid" || undefined,
   135→        noshow: kind === "noshow" || undefined,

   143→        duration: 3000,
   144→      });
   145→    } else if (kind === "unpaid") {
   146→      toast({
   147→        title: `Не оплачено: ${client.name}`,
   148→        description: "Долг зафиксирован — попадёт в раздел «Долги».",
   149→        duration: 3200,
   150→      });
   151→    } else {
   152→      toast({
   153→        title: `Клиента нет: ${client.name}`,
   154→        description: "Точка закрыта, вернёмся в следующий заезд.",
   155→        duration: 3200,
   156→      });

   159→
   160→  /* ── Пустое состояние ── */

   163→      <div className="mx-auto w-full max-w-7xl px-4 pb-8 sm:px-6 lg:px-8">

   256→            <Minus className="size-4" />
   257→          </button>

   289→                  "flex flex-wrap items-center gap-3 rounded-2xl border bg-surface p-4 sm:flex-nowrap",

   325→                    <span className="flex items-center gap-1 text-ink-soft">
   326→                      <Clock className="size-3" />

   344→                    onClick={() =>

   347→                    aria-label="Минус коробка"
   348→                    className="flex size-8 items-center justify-center rounded-lg border border-border text-ink transition-colors hover:bg-tint"

   373→                  >

   380→        </ol>

   382→        {/* Не вместилось */}

   384→          <div className="mt-4">

   438→          className="inline-flex h-11 items-center gap-2 rounded-xl bg-accent px-5 text-sm font-semibold text-[#1C1A17] transition-colors hover:bg-accent/90"

   440→          <Navigation className="size-4" />
   441→          Поехать

   448→/* ── Заголовок страницы ─────────────────────────────────────── */
   449→
   450→function RouteHeader({ points }: { points: number }) {
   451→  const [today, setToday] = useState<Date | null>(null);
   452→  useEffect(() => {
   453→    const id = requestAnimationFrame(() => setToday(new Date()));
   454→    return () => cancelAnimationFrame(id);
   455→  }, []);
   456→  return (
   457→    <header className="flex flex-wrap items-center justify-between gap-3 py-6 lg:py-8">
   458→      <div className="min-w-0">
   459→        <h1 className="flex items-center gap-2.5 text-xl font-semibold tracking-tight text-ink sm:text-2xl">
   460→          <RouteIcon className="size-6 text-accent" />
   461→          Маршрут на день
   462→        </h1>
   463→        <p className="mt-0.5 text-sm capitalize text-ink-soft">
   464→          {today
   465→            ? format(today, "EEEE, d MMMM", { locale: ru })
   466→            : "\u00A0"}
   467→          {points > 0 ? ` · ${points} точек` : ""}
   468→        </p>
   469→      </div>
   470→    </header>
   471→  );
   472→}

   474→/* ── Режим «В дороге» ───────────────────────────────────────── */
   475→
   476→function DrivingMode({
   477→  currentStop,
   478→  stopsTotal,
   479→  doneCount,
   480→  notes,
   481→  setNotes,
   482→  onMark,
   483→  onExit,
   484→}: {
   485→  currentStop: {
   486→    client: Client;
   487→    boxes: number;
   488→    index: number;
   489→    etaMinutes: number;
   490→  } | null;
   491→  stopsTotal: number;
   492→  doneCount: number;
   493→  notes: VoiceNote[];
   494→  setNotes: React.Dispatch<React.SetStateAction<VoiceNote[]>>;
   495→  onMark: (client: Client, kind: "delivered" | "unpaid" | "noshow", boxes: number) => void;
   496→  onExit: () => void;
   497→}) {
   498→  const { toast } = useToast();
   499→  const recorderRef = useRef<MediaRecorder | null>(null);
   500→  const streamRef = useRef<MediaStream | null>(null);
   501→  const [recording, setRecording] = useState(false);
   502→  const [seconds, setSeconds] = useState(0);
   503→  const startedAtRef = useRef(0);
   504→
   505→  // Таймер записи.
   506→  useEffect(() => {
   507→    if (!recording) return;
   508→    const t = setInterval(() => setSeconds((s) => s + 1), 1000);
   509→    return () => clearInterval(t);
   510→  }, [recording]);
   511→
   512→  // Освобождаем микрофон при размонтировании.
   513→  useEffect(
   514→    () => () => {
   515→      streamRef.current?.getTracks().forEach((t) => t.stop());
   516→    },
   517→    []
   518→  );
   519→
   520→  const startRecording = async () => {
   521→    if (typeof MediaRecorder === "undefined" || !navigator.mediaDevices?.getUserMedia) {
   522→      toast({
   523→        title: "Микрофон недоступен",
   524→        description: "Браузер запретил запись — заметку можно набрать текстом.",
   525→        duration: 3400,
   526→      });
   527→      return;
   528→    }
   529→    try {
   530→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   531→      streamRef.current = stream;
   532→      const mr = new MediaRecorder(stream);
   533→      const chunks: Blob[] = [];
   534→      mr.ondataavailable = (e) => chunks.push(e.data);
   535→      mr.onstop = () => {
   536→        const blob = new Blob(chunks, { type: mr.mimeType || "audio/webm" });
   537→        const url = URL.createObjectURL(blob);
   538→        const elapsed = Math.max(
   539→          1,
   540→          Math.round((Date.now() - startedAtRef.current) / 1000)
   541→        );
   542→        setNotes((n) => [
   543→          {
   544→            id: `note-${Date.now()}`,
   545→            url,
   546→            seconds: elapsed,
   547→            stopName: currentStop?.client.name ?? "По маршруту",
   548→            at: new Date(),
   549→          },
   550→          ...n,
   551→        ]);
   552→        toast({
   553→          title: "Голосовая заметка сохранена",
   554→          description: `${elapsed} сек · ${currentStop?.client.name ?? "маршрут"}`,
   555→          duration: 2400,
   556→        });
   557→      };
   558→      recorderRef.current = mr;
   559→      startedAtRef.current = Date.now();
   560→      setSeconds(0);
   561→      mr.start();
   562→      setRecording(true);
   563→    } catch {
   564→      toast({
   565→        title: "Микрофон недоступен",
   566→        description: "Разрешите доступ к микрофону в браузере.",
   567→        duration: 3400,
   568→      });
   569→    }
   570→  };
   571→
   572→  const stopRecording = () => {
   573→    recorderRef.current?.stop();
   574→    streamRef.current?.getTracks().forEach((t) => t.stop());
   575→    setRecording(false);
   576→  };
   577→
   578→  return (
   579→    <div className="mx-auto flex min-h-[calc(100dvh-96px)] w-full max-w-3xl flex-col px-4 pb-8 sm:px-6 lg:min-h-dvh">
   580→      <header className="flex items-center justify-between gap-3 py-5">
   581→        <div>
   582→          <h1 className="flex items-center gap-2 text-xl font-semibold tracking-tight text-ink">
   583→            <Truck className="size-6 text-accent" />
   584→            В дороге
   585→          </h1>
   586→          <p className="mt-0.5 text-sm text-ink-soft">
   587→            Точка{" "}
   588→            <span className="font-mono tabular-nums">
   589→              {Math.min(doneCount + 1, stopsTotal)}/{stopsTotal}
   590→            </span>{" "}
   591→            закрыто <span className="font-mono tabular-nums">{doneCount}</span>
   592→          </p>
   593→        </div>
   594→        <button
   595→          type="button"
   596→          onClick={onExit}
   597→          className="inline-flex h-10 items-center gap-1.5 rounded-xl border border-border bg-surface px-4 text-sm font-medium text-ink transition-colors hover:bg-tint"
   598→        >
   599→          <X className="size-4" />
   600→          Выйти
   601→        </button>
   602→      </header>
   603→
   604→      {currentStop ? (
   605→        <>
   606→          {/* Текущая точка */}
   607→          <section className="rounded-2xl border border-border bg-surface p-5">
   608→            <div className="flex items-start justify-between gap-3">
   609→              <div className="min-w-0">
   610→                <p className="text-[11px] font-semibold uppercase tracking-wide text-ink-soft">
   611→                  Сейчас на точке
   612→                </p>
   613→                <h2 className="mt-1 text-xl font-bold tracking-tight text-ink">
   614→                  {currentStop.client.name}
   615→                </h2>
   616→                <p className="mt-1 text-[13px] text-ink-soft">
   617→                  {currentStop.client.address} · {currentStop.client.district}
   618→                </p>
   619→              </div>
   620→              <div className="shrink-0 rounded-xl bg-tint px-3 py-2 text-center">
   621→                <p className="font-mono text-lg font-bold tabular-nums leading-none text-ink">
   622→                  {currentStop.boxes}
   623→                </p>
   624→                <p className="mt-1 text-[10px] uppercase tracking-wide text-ink-soft">
   625→                  коробок
   626→                </p>
   627→              </div>
   628→            </div>
   629→          </section>
   630→
   631→          {/* Большие кнопки результата */}
   632→          <section className="mt-4 grid grid-cols-1 gap-3 sm:grid-cols-3">
   633→            <BigAction
   634→              label="Доставлено"
   635→              icon={<Package className="size-7" />}
   636→              onClick={() => onMark(currentStop.client, "delivered", currentStop.boxes)}
   637→              fg="var(--tier-green)"
   638→            />
   639→            <BigAction
   640→              label="Не оплачено"
   641→              icon={<Wallet className="size-7" />}
   642→              onClick={() => onMark(currentStop.client, "unpaid", currentStop.boxes)}
   643→              fg="var(--tier-yellow)"
   644→            />
   645→            <BigAction
   646→              label="Клиента нет"
   647→              icon={<UserX className="size-7" />}
   648→              onClick={() => onMark(currentStop.client, "noshow", currentStop.boxes)}
   649→              fg="var(--tier-red)"
   650→            />
   651→          </section>
   652→        </>
   653→      ) : (
   654→        <section className="flex flex-col items-center gap-3 rounded-2xl border border-border bg-surface py-14 text-center">
   655→          <Package className="size-10" style={{ color: "var(--tier-green)" }} />
   656→          <p className="text-[16px] font-semibold text-ink">Все точки закрыты</p>
   657→          <p className="max-w-xs text-sm text-ink-soft">
   658→            Маршрут выполнен. Возвращайтесь на склад — вторая ходка уже
   659→            ждёт на экране маршрута.
   660→          </p>
   661→        </section>
   662→      )}
   663→
   664→      {/* Голосовая заметка */}
   665→      <section className="mt-4 rounded-2xl border border-border bg-surface p-4">
   666→        <div className="flex items-center justify-between gap-3">
   667→          <div>
   668→            <p className="text-[14px] font-semibold text-ink">Голосовая заметка</p>
   669→            <p className="text-[12px] text-ink-soft">
   670→              Продиктуйте детали по точке — прослушаете на складе
   671→            </p>
   672→          </div>
   673→          <button
   674→            type="button"
   675→            onClick={recording ? stopRecording : startRecording}
   676→            aria-label={recording ? "Остановить запись" : "Записать заметку"}
   677→            className={cn(
   678→              "relative flex size-14 items-center justify-center rounded-full text-white shadow-lg transition-transform active:scale-95",
   679→              recording && "animate-pulse"
   680→            )}
   681→            style={{
   682→              background: recording ? "var(--tier-red)" : "var(--accent)",
   683→              color: recording ? "#fff" : "#1C1A17",
   684→            }}
   685→          >
   686→            {recording ? <Square className="size-5" /> : <Mic className="size-6" />}
   687→          </button>
   688→        </div>
   689→        {recording ? (
   690→          <p className="mt-2 text-center font-mono text-[13px] tabular-nums text-ink">
   691→            ● Запись… {String(Math.floor(seconds / 60)).padStart(2, "0")}:
   692→            {String(seconds % 60).padStart(2, "0")}
   693→          </p>
   694→        ) : null}
   695→
   696→        {notes.length > 0 ? (
   697→          <ul className="mt-3 space-y-2">
   698→            {notes.map((n) => (
   699→              <li
   700→                key={n.id}
   701→                className="flex items-center gap-3 rounded-xl border border-border px-3 py-2"
   702→              >
   703→                <Play className="size-4 shrink-0 text-ink-soft" />
   704→                <div className="min-w-0 flex-1">
   705→                  <p className="truncate text-[13px] font-medium text-ink">
   706→                    {n.stopName}
   707→                  </p>
   708→                  <p className="text-[11px] text-ink-soft">
   709→                    {n.at.toLocaleTimeString("ru-RU", {
   710→                      hour: "2-digit",
   711→                      minute: "2-digit",
   712→                    })}{" "}
   713→                    · {n.seconds} сек
   714→                  </p>
   715→                </div>
   716→                <audio controls preload="none" src={n.url} className="h-8 max-w-[45%]" />
   717→              </li>
   718→            ))}
   719→          </ul>
   720→        ) : (
   721→          <p className="mt-3 flex items-center gap-1.5 text-[12px] text-ink-soft">
   722→            <MicOff className="size-3.5" />
   723→            Заметок пока нет
   724→          </p>
   725→        )}
   726→      </section>
   727→    </div>
   728→  );
   729→}
   730→
   731→function BigAction({
   732→  label,
   733→  icon,
   734→  onClick,
   735→  fg,
   736→}: {
   737→  label: string;
   738→  icon: React.ReactNode;
   739→  onClick: () => void;
   740→  fg: string;
   741→}) {
   742→  return (
   743→    <button
   744→      type="button"
   745→      onClick={onClick}
   746→      className="flex min-h-[92px] flex-col items-center justify-center gap-2 rounded-2xl border font-semibold transition-transform active:scale-[0.98]"
   747→      style={{
   748→        color: fg,
   749→        borderColor: `color-mix(in srgb, ${fg} 45%, transparent)`,
   750→        background: `color-mix(in srgb, ${fg} 10%, transparent)`,
   751→      }}
   752→    >
   753→      {icon}
   754→      <span className="text-[15px]">{label}</span>
   755→    </button>
   756→  );
   757→}
